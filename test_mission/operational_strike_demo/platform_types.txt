# Generic, cooperative platform types for the network-resource demonstration.
# No platform carries a weapon and no opposing side is created.

platform_type NRM_OP_GROUND_BASE WSF_PLATFORM
   add processor message_processor NRM_OP_MESSAGE_PROCESSOR end_processor
end_platform_type

platform_type NRM_OP_LINK11_GROUND NRM_OP_GROUND_BASE
   side blue
   add comm link11 NRM_OP_LINK11_COMM
      internal_link message_processor
   end_comm
end_platform_type
