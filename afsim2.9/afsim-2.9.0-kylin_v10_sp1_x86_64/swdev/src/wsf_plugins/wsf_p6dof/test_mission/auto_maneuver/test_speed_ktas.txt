include_once ../platforms/fa-lgt.txt
include_once ../scripts/utils.txt

end_time 600 s

platform TEST_AIRCRAFT FA-LGT
   side blue
   
   route
      position   21.325n   158.000w  altitude   20000.0 ft   speed  450.0 kts
      position   20.00n    157.98w   altitude   20000.0 ft   speed  450.0 kts
   end_route
   
   script_variables
      double initialSpeed;
      bool allPassed = true;        
   end_script_variables
   
   execute at_time 10 sec absolute
      WsfSpeedKTAS_Maneuver mvr = WsfSpeedKTAS_Maneuver.Construct(350.0);
      mvr.SetEntryConstraint(WsfManeuverConstraint.AT_RELATIVE_TIME(10.0));
      mvr.SetExitConstraint(WsfManeuverConstraint.AT_RELATIVE_TIME(3.0));
      
      allPassed = ExpectDouble(mvr.GetSpeed(), 350.0, 1.0e-4, "KTAS speed should be 350.0") && allPassed;

      initialSpeed = PLATFORM.Speed();

      WsfP6DOF_Mover mov = (WsfP6DOF_Mover)PLATFORM.Mover();      
      mov.ExecuteManeuver(mvr);
   end_execute
   
   execute at_time 12 sec absolute
      allPassed = ExpectDouble(PLATFORM.Speed(), initialSpeed, 1.0, "Speed should not have changed") && allPassed;
   end_execute
   
   execute at_time 19.9 sec absolute
      allPassed = ExpectDouble(PLATFORM.Speed(), initialSpeed, 1.0, "Speed should not have changed during pending") && allPassed;
   end_execute
   
   execute at_time 300 sec absolute
      allPassed = ExpectDouble(PLATFORM.Speed(), 180.0, 1.0, "Speed should match.") && allPassed;
   end_execute
   
   execute at_time 400 s absolute
      if (allPassed)
      {
         writeln("-PASS-");
      }
      else
      {
         writeln("-FAIL- One or more tests failed.");
      }   
   end_execute      
end_platform
